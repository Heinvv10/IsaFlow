// Not run by CI.
import { act, cleanup, fireEvent, render, screen, waitFor } from '@testing-library/react';
import { describe, expect, it, vi } from 'vitest';
import { BudgetWorkspace } from '@/components/accounting/budgets/workspace/BudgetWorkspace';
import { buildLayout } from '@/components/accounting/budgets/workspace/gridLayout';
import { ACTUAL_REASON, GridStore } from '@/components/accounting/budgets/workspace/gridStore';
import { installNetwork, isModel, ok } from '../workspace/workspaceHarness';
import { ID, MONTHS, driverModel } from './driverFixture';

vi.mock('@/contexts/CompanyContext', () => ({ useCompany: () => ({ companyRole: 'owner' }) }));
vi.mock('next/router', () => ({ useRouter: () => ({ events: { on: vi.fn(), off: vi.fn(), emit: vi.fn() }, push: vi.fn() }) }));
const rowOf = (id: string) => buildLayout(driverModel().accounts, MONTHS).rows.find((r) => r.key === `acct:${id}`)!.index;
const cellAt = (id: string, c: number) => document.querySelector<HTMLElement>(`[data-pos="${rowOf(id)}:${c}"]`)!;

const log: string[] = [];
const ids = new WeakMap<object, number>(); let nextId = 0;
const idOf = (o: object) => { if (!ids.has(o)) ids.set(o, ++nextId); return ids.get(o)!; };
const proto = GridStore.prototype as unknown as Record<string, (...a: unknown[]) => unknown>;
let firstListeners = -1;
let lastStore: { metaListeners?: Set<unknown>; getMeta?: () => { announcement: { text: string } } } | null = null;
const tally = new Map<string, number>();
for (const name of ['announce', 'reset', 'setGridState', 'setPrior', 'setAccess']) {
  const orig = proto[name]!;
  proto[name] = function (this: unknown, ...a: unknown[]) {
    if (name === 'setGridState' && firstListeners === -1) firstListeners = ((this as { metaListeners?: Set<unknown> }).metaListeners?.size) ?? -2;
    lastStore = this as typeof lastStore;
    const me = this as { metaListeners?: Set<unknown>; meta?: { announcement: { seq: number; text: string } } };
    if (name === 'announce' || name === 'reset' || name === 'setPrior') log.push(`#${idOf(this as object)}[listeners=${me.metaListeners?.size}]`);
    log.push(name === 'setGridState' ? `${name}:${JSON.stringify((a[0] as { focus: unknown; mode: string }).focus)}/${(a[0] as { mode: string }).mode}` : name === 'announce' ? `announce:${String(a[0]).slice(0, 12)}` : name);
    return orig.apply(this, a);
  };
}
it('loop', async () => {
  let bad = 0;
  const N = Number(process.env.N ?? 150);
  for (let i = 0; i < N; i++) {
    const net = installNetwork();
    net.on(isModel, () => ok(driverModel({ withActuals: true })));
    (globalThis as { __bg?: string[] }).__bg = [];
    render(<BudgetWorkspace />);
    await screen.findByRole('columnheader', { name: 'March 2026, actual' });
    const mounted = () => ((globalThis as { __bg?: string[] }).__bg ?? []).includes('effect-mounted');
    if (process.env.FIXMODE === 'nudge') {
      let tries = 0;
      while (!mounted() && tries++ < 20) { act(() => {}); await act(async () => {}); }
      if (tries > 1) process.stdout.write(`NUDGED i=${i} tries=${tries}\n`);
    }
    if (!mounted()) process.stdout.write(`NOTMOUNTED i=${i}\n`);
    const mar = cellAt(ID.rev1, 0);
    log.length = 0; firstListeners = -1;
    log.push(`pos=${mar.getAttribute('data-pos')}`);
    fireEvent.mouseDown(mar);
    log.push('--mouseDown done');
    fireEvent.keyDown(mar, { key: '7' });
    const read = () => screen.getByTestId('grid-announcement').textContent;
    let got = read();
    if (process.env.FIXMODE === 'waitfor') { try { await waitFor(() => expect(read()).toBe(ACTUAL_REASON)); } catch { /* reported below */ } got = read(); }
    if (got !== ACTUAL_REASON) {
      bad++;
      process.stdout.write(`LOOPFAIL i=${i} got=${JSON.stringify(got)} connected=${mar.isConnected} same=${cellAt(ID.rev1, 0) === mar} listenersAfter=${lastStore?.metaListeners?.size} metaText=${JSON.stringify(lastStore?.getMeta?.().announcement.text.slice(0,10))} annEls=${document.querySelectorAll('[data-testid="grid-announcement"]').length} bg=${JSON.stringify((globalThis as { __bg?: string[] }).__bg)} sent=${net.sent.length} stores=${nextId} LOG=${JSON.stringify(log)} active=${document.activeElement?.tagName}\n`);
    }
    { const k = `listeners@mouseDown=${firstListeners} ok=${got === ACTUAL_REASON}`; tally.set(k, (tally.get(k) ?? 0) + 1); }
    cleanup();
    vi.unstubAllGlobals();
  }
  process.stdout.write(`LOOPDONE bad=${bad}/${N} tally=${JSON.stringify([...tally])}\n`);
  expect(bad).toBe(0);
}, 600_000);
